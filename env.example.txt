# ========================================
# 環境變數設定範例
# ========================================
# 
# 在 Railway.app 部署時，請在 Variables 設定中新增以下變數：
#

# 資料庫配置（從 Railway MySQL 服務複製）
DB_HOST=your-mysql-host-from-railway
DB_USERNAME=root
DB_PASSWORD=your-mysql-password-from-railway
DB_NAME=railway
DB_PORT=3306

# 或使用 Railway 自動提供的變數（推薦）
MYSQL_HOST=your-mysql-host
MYSQL_USER=root
MYSQL_PASSWORD=your-password
MYSQL_DATABASE=railway
MYSQL_PORT=3306

# JWT 密鑰（請更改為隨機字串，至少 32 字元）
JWT_SECRET=請更改為一個超級安全的隨機密鑰字串至少32字元長度

# 應用配置
APP_ENV=production
APP_DEBUG=false
APP_URL=https://your-app.up.railway.app

# 時區
TIMEZONE=Asia/Taipei

# ========================================
# 注意事項
# ========================================
# 1. 不要直接使用這個檔案！請在 Railway Dashboard 中設定環境變數
# 2. JWT_SECRET 必須是隨機且安全的字串
# 3. DB_* 變數請從 Railway MySQL 服務的 Variables 標籤複製
# 4. APP_URL 請在取得 Railway 網址後更新
