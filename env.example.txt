# 郵件設定 (Email Configuration)

# SMTP 主機 (Gmail 為 smtp.gmail.com)
SMTP_HOST=smtp.gmail.com

# SMTP 使用者名稱 (通常是您的 Gmail 地址)
SMTP_USERNAME=portfolioplus2025@gmail.com

# SMTP 密碼 (Gmail 請使用「應用程式密碼」)
# 如何取得 Gmail 應用程式密碼：
# 1. 前往 Google 帳戶設定 (myaccount.google.com)
# 2. 選擇「安全性」
# 3. 啟用「兩步驟驗證」
# 4. 在「應用程式密碼」中產生新密碼
SMTP_PASSWORD=your_app_password_here

# SMTP 連接埠 (Gmail TLS 使用 587)
SMTP_PORT=587

# 聯絡表單收件人信箱 (接收聯絡訊息的信箱)
CONTACT_EMAIL=portfolioplus2025@gmail.com
